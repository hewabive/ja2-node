# Our mercs off the loaded map between hourly updates (0029). Every block starts from a fresh save load and sets the
# JA2 clock, the stamp of the last real-time turn and the rotting corpse stamp with abclock, the screen with abscreen
# and the compression with abcompress, so nothing carries over from the save's screen or an earlier block. Every block
# ends on the game screen: LoadSavedGame crashes after a death on the map while the map screen is up.
# SaveGame01: B13 at 21:37 (night), Shadow 0, Sidney 1, Ira 2 (profile 59); civilians 116-123.

# 1. The real-time turn on the game screen: Ira off to B12, Sidney wounded on the map.
save SaveGame01
abclock ja2clock=6050930 last=6050930 rotting=0 base=1
abscreen screen=game
abcompress mode=1 on=0 paused=0
abturn                                              # the first turn: looks, opplists, everyone in B13
abaway who=2 sector=B12                             # RemoveSoldierFromTacticalSector: NOWHERE
abset who=2 life=50 bleeding=20 nextBleed=0.5 breath=40 breathRed=3000
abset who=1 life=60 bleeding=20 nextBleed=0.5
abturn                                              # id order: Sidney's bleeding line, then Ira's ME_TOO
abturn count=3
abset who=2 anim=RUNNING breath=40                  # the animation he left the map in decides his breath
abturn
abset who=2 anim=STANDING terrain=10                # standing in deep water (MercInWater reads his own terrain)
abturn
abset who=2 terrain=5 gassed=1                      # the gas test at NOWHERE
abturn
abset who=0 gassed=1
absmoke grid=6634 item=SMOKE_GRENADE                # a cloud on Shadow's tile keeps him gassed
abturn
abset who=2 drug=0:0:-2:2:0:20 regen=3 morale=50    # an adrenaline crash and the regeneration booster off the map
abturn count=2
abset who=0 buddy=59
abset who=2 life=1 bleeding=5 nextBleed=0.5 regen=0 insurance=1 insuredAt=0
abturn                                              # Ira bleeds to death in B12: no body, no buddy line
abrecords sector=B12
abscreen screen=game

# 2. The map screen's poll: gate and check time per compression mode, the 20 ms counter satisfied by sleeping.
save SaveGame01
abclock ja2clock=1000000 last=1000000 rotting=0 base=1
abscreen screen=map
abaway who=2 sector=B12
abset who=2 life=50 bleeding=20 nextBleed=2
abcompress mode=0 on=0 paused=0
abpoll now=1000020 count=2                          # X0: nothing
abcompress mode=2 on=1 paused=0
abpoll now=1000060 count=3                          # 5 min/s: check 1 ms, a turn at each poll
abpoll now=1000121 count=2 step=1                   # 1 ms apart: not past the check
abcompress mode=3 on=1 paused=0
abpoll now=1000200 count=2 step=1                   # 30 min/s: check 0
abcompress mode=4 on=1 paused=0
abpoll now=1000300 count=2 step=1                   # 60 min/s: check 0
abcompress mode=2 on=1 paused=1
abpoll now=1000400 count=2                          # paused: nothing
abcompress mode=2 on=0 paused=0
abpoll now=1000500 count=2                          # compression stopped: nothing
abscreen screen=game
abcompress mode=1 on=0 paused=0
abpoll now=1005301 count=2 step=1                   # the game screen: 5000 ms, past it only at the second poll
abscreen screen=map
abcompress mode=2 on=1 paused=0
abclock decay=162750
abpoll now=1010000                                  # blood and smells: 5 real seconds at 5 min/s, 1500 s not past
abclock decay=162749
abpoll now=1010020
abcompress mode=3 on=1 paused=0
abclock decay=155250
abpoll now=1010040
abclock decay=155249
abpoll now=1010060
abcompress mode=4 on=1 paused=0
abclock decay=146250
abpoll now=1010080
abclock decay=146249
abpoll now=1010100
abscreen screen=game
abcompress mode=1 on=0 paused=0
abclock decay=164150
abpoll now=1016000
abclock decay=164149
abpoll now=1022000
abscreen screen=game

# 3. Deaths on the map screen: Ira in B12, then Sidney between B12 and the next sector.

# 4. A merc of the loaded sector dying on the map: collapsed (death animation, the map body in B13's records), standing
# (dead in tactical terms, the map branch skipped), then the world unloaded.

# 6. Sight returning on the map reveals a hidden room with an item; a light step of dusk on the map.
save SaveGame01
abclock ja2clock=5000000 last=5000000 rotting=0 base=1
abscreen screen=game
abcompress mode=1 on=0 paused=0
abturn
abscreen screen=map

abset who=1 grid=10433 blinded=1                    # Sidney inside room 1, blind for one more turn
abhide room=1 item=GLOCK_17 at=10595
abturn                                              # his look reveals the room: the spotted-something line
ablight value=3                                     # EVENT_CHANGELIGHTVAL on the map: the loaded map's ambient stays
abturn
abscreen screen=game
