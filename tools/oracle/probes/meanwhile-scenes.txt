# Palace meanwhile scenes played frame by frame on the virtual clock (0030, 0031, 0032): the event, the box (answered OK
# or Skip through its callback, as the message box screen does), the fade out, the load of P3_m, the fade in, the Queen's
# records with their quotes, actions and walks until END_MEANWHILE, the fade out, the way back and the fade in. Each block
# loads its save with the draws traced (the import's restart takes them), reseeds the engine and the PreRandom ring (the
# saves hold an all-zero ring) and stops at each phase: the box up, the hall loaded, the scene under way, the scene ended,
# the old screen back. With the virtual clock and the seeds a run is byte for byte the same every time.

hdtimer on=1
seed value=3

# The outskirts scene started on the game screen in B13: Elliot shot, Joe walks off, the way back reloads B13.
draws on
save SaveGame01
draws dump
seed value=7 pre=reseed
mwset screen=game seen=0
meanwhilepost kind=outskirts
mwbegin
mwframes count=2000 until=box
mwframes count=2000 answer=ok until=loaded
mwframes count=2000 until=scene
mwframes count=30000 until=ended
mwframes count=2000 until=back

# Drassen liberated started on the map screen with 30-minute compression on and A9 selected: the map screen's entry
# runs in the first frame; the scene stops compression, and the way back restores the selection before the map
# screen's entry selects the loaded sector again.
draws on
save SaveGame01
draws dump
seed value=11 pre=reseed
mwset screen=map seen=0
mwframes count=1
mwset compress=3 select=A9
meanwhilepost kind=town param=2
mwbegin
mwframes count=2000 until=box
mwframes count=2000 answer=ok until=loaded
mwframes count=2000 until=scene
mwframes count=30000 until=ended
mwframes count=2000 until=back
