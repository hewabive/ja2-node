# The helicopter drop and the helicopter's passengers over the loaded sector (0031, README "Helicopter drop").
# hdtimer on=1 switches 0030's virtual clock on before the first load. Every drop block starts a new game the way the
# main menu does after a game (hdrestart: ReStartingGame), hires through 0010 and runs the map screen's start
# (hdjackedin: HandleTimeCompressWithTeamJackedInAndGearedToGo, with A9 and its enemies loaded); hdframes then runs the
# game screen's frames on the virtual clock, 5 ms each. hdset live=1 lets the fades ramp as with a sample playing.
# Every new game is seeded the same, so each block starts from the same profiles and A9 placements, and the AI's draws
# repeat on every run.

# 1. The first drop of a new game: three mercs, one round, enemies in A9, the end of the first run (turn times, the
# enemies called to the sweet spot, the enemy presence quote, the music), then the step-aways and the AI after it.
hdtimer on=1
seed value=21 pre=reseed
newgamefull difficulty=2
setbalance amount=500000
aimhire profile=15 contract=7 gear=0
aimhire profile=4 contract=7 gear=0
aimhire profile=20 contract=7 gear=0
hdset live=1
hdjackedin
hdframes count=20000 until=end
hdframes count=1000

# Passengers of the helicopter over the loaded sector, on SaveGame01 (B13, the helicopter's base; Shadow 0, Sidney 1,
# Ira 2, profile 59). hpsetup gives the player the helicopter in B13, hpboard boards in tactical (PutSoldierInVehicle).

# 11. Boarding, the turn starts out of and in combat (set with an enemy four tiles from Ira, so it goes on), a nervous
# walker next to the passengers and a nervous passenger, quotes from anybody with and without a walker to say them.
save SaveGame01
hpset screen=game
hpsetup sector=B13
hpboard who=0
hpboard who=1
hpturn
hpenemy grid=9675 dir=0
hpset enemyInSector=1 combat=1
hpturn
hpturn
hpset combat=0
hpset who=2 personality=nervous morale=40
hpturn
hpset who=0 personality=nervous morale=40
hpturn
hpquote quote=59
hpquote quote=59
hpset who=2 asleep=1
hpquote quote=59

# 12. Damage to a passenger, his death over the loaded sector and the frames after it.
save SaveGame01
hpset screen=game
hpsetup sector=B13
hpboard who=0
hpdamage who=0 damage=10 reason=gunfire
hpdamage who=0 damage=100 reason=gunfire
hdframes count=600
hpstate
